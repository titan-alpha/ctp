## ConveniencePro Tool Protocol (CTP) Compliance

### Protocol Implementation

All 385 tools implement the **ConveniencePro Tool Protocol (CTP)**, an open specification for:

1. **HTTP API Access** - RESTful endpoints for programmatic use
2. **Embeddable Widgets** - Iframe-based integration for any website
3. **AI Assistant Integration** - Model Context Protocol (MCP) compatibility

### CTP Pattern

Every tool follows this standardized pattern:

```typescript
import { success, failure } from '@conveniencepro/ctp-core'
import type { ToolResult } from '@conveniencepro/ctp-core'

// 1. Type-safe parameter interface
export interface ToolParams {
  required: string
  optional?: number
}

// 2. Type-safe result interface
export interface ToolResultData extends Record<string, unknown> {
  output: string
  metadata: {
    processingTime: number
  }
}

// 3. Tool implementation
export function toolName(params: ToolParams): ToolResult<ToolResultData> {
  // Validation
  if (!params.required) {
    return failure('Required parameter missing', 'MISSING_REQUIRED')
  }

  // Processing
  try {
    const output = processInput(params.required)

    return success<ToolResultData>({
      output,
      metadata: {
        processingTime: Date.now() - start
      }
    })
  } catch (error) {
    return failure(
      error instanceof Error ? error.message : 'Unknown error',
      'EXECUTION_ERROR'
    )
  }
}

// 4. Default export
export default toolName
```
